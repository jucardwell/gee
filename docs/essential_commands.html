4.2 Filtering Image Collections

4.2.1 Filter by a Date Range

// filter by a date
var filtered_collection = image_collection.filterDate('2021-08-01', '2021-08-02');

//filter by a set of dates
var filtered_collection = image_collection.filterDate('2021-08-01', '2022-08-02');

//filter by a year or set of years
var filtered_collection = image_collection.filter(
    ee.Filter.calendarRange(2021, 2023, 'year')
);

//filter by a month
var filtered_collection = image_collection.filter(
    ee.Filter.calendarRange(8, 8, 'month')
);

//filter by day of year
var filtered_collection = image_collection.filter(
    ee.Filter.dayOfYear(150, 200)
);

//filter by day of the month
var filtered_collection = image_collection.filter(
    ee.Filter.calendarRange(1, 15, 'day_of_month')
);

4.2.2 Filter by a Location

Filter an image collection to images that intersect a geometry.

// Filter by a point (coordinates)
var point = ee.Geometry.Point([-122.292, 37.901]);

var filtered_collection = image_collection.filterBounds(point);
// Filter by an ee.Geometry, ee.Feature, or ee.FeatureCollection
//in this case roi needs to be an ee.Geometry or ee.FeatureCollection
var filtered_collection = image_collection.filterBounds(roi);

4.2.3 Filter by Metadata Properties

For these, you need to reference the actual metadata to see what the variables are called

//cloud cover less than 10%
var filtered_collection = image_collection.filter(
    ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 10)
);

//cloud cover greater than 10%
var filtered_collection = image_collection.filter(
    ee.Filter.gt('CLOUDY_PIXEL_PERCENTAGE', 10)
);


//cloud cover equal to 10%
var filtered_collection = image_collection.filter(
    ee.Filter.eq('CLOUDY_PIXEL_PERCENTAGE', 10)
);

//cloud cover between 10% and 20%
var filtered_collection = image_collection.filter(
    ee.Filter.and(
        ee.Filter.gte('CLOUDY_PIXEL_PERCENTAGE', 10),
        ee.Filter.lte('CLOUDY_PIXEL_PERCENTAGE', 20)
    )
);

4.3.3 Visualizing an Image Collection

The Map.addLayer command expects a single image. To visualize an image collection you need to reduce it to a single image. The easiest way to do this is to take the median value for each pixel across the image collection. You can also use mean(), max(), or min() to reduce the image collection.


Map.addLayer(
    image_collection.median(), //takes the median of the image collection
    {
        bands: ['SR_B3', 'SR_B2', 'SR_B1'], //bands will automatically fill RGB
        min: min,
        max: max
    },
    'Layer Name' //name to show in Layer Manager
);

4.3.4 Help Picking Min/Max

An easy way to pick min/max values is to take a sample of the raster and use that to see percentile values


// for an image
var stats = image.select('band')
  .sample({
    region: image.geometry(),
    numPixels: 1000,
    scale: 30
  })
  .reduceColumns({
    reducer: ee.Reducer.percentile([2, 98]),
    selectors: ['band']
  });

print(stats);


// for an image collection
var stats = image_collection.median().select('band')
  .sample({
    region: image_collection.first().geometry(),
    numPixels: 1000,
    scale: 30
  })
  .reduceColumns({
    reducer: ee.Reducer.percentile([2, 98]),
    selectors: ['band']
  });

print(stats);

4.5 Band Math

Band math allows you to create new raster bands by mathematically combining existing bands. Common operations include addition, subtraction, multiplication, division, and normalized differences.

4.5.1 Basic Mathematical Operations

// select individual bands 

var band_1 = image.select('band_1'); 
var band_2 = image.select('band_2'); 

// add two bands 
var added = band_1.add(band_2); 

// subtract
var subtracted = band_1.subtract(band_2);

// multiply two bands 
var multiplied = band_1.multiply(band_2); 

// divide 
var divided = band_1.divide(band_2);

//built in normalized difference function

var normalized_difference = image.normalizedDifference(['band_1', 'band_2'])

4.5.2 Using Expressions

Expressions can be used for more complicated operations


var calculated_band = image.expression( 
                    '(BAND_1 - BAND_2) / (BAND_1 + BAND_2)', 
                    { BAND_1: image.select('band_1'), 
                    BAND_2: image.select('band_2') });

4.6.4 Cloud Masking


//Landsat 8/9 Collection 2 Level 2 
function maskLandsatClouds(image) {
  // Select the QA band that stores per-pixel quality flags
  var qa = image.select('QA_PIXEL');
  // Bit 1 = dilated cloud, 2 = cirrus, 3 = cloud, 4 = cloud shadow
  var badBits = (1 << 1) | (1 << 2) | (1 << 3) | (1 << 4);
  // A pixel is clear if NONE of those bits are turned on
  var clear = qa.bitwiseAnd(badBits).eq(0);
  // Keep the clear pixels, drop everything else
  return image.updateMask(clear);
}

// Sentinel-2 SR Harmonized, masked with Cloud Score+
function maskS2clouds(image) {
  // Look up this image's matching Cloud Score+ image by its ID
  var csPlus = ee.ImageCollection('GOOGLE/CLOUD_SCORE_PLUS/V1/S2_HARMONIZED');
  var match = csPlus.filter(
    ee.Filter.eq('system:index', image.get('system:index'))).first();
  // Scores run 0 (occluded) to 1 (clear)
  var score = ee.Image(match).select('cs_cdf');
  // A pixel is clear if its score is at or above the threshold
  var clear = score.gte(0.60);
  // Keep the clear pixels, drop everything else
  return image.updateMask(clear);
}

    var evi = image.expression(
        '2.5 * ((NIR - RED) / (NIR + 6 * RED - 7.5 * BLUE + 1))',
        {
            NIR: image.select('B8'),
            RED: image.select('B4'),
            BLUE: image.select('B2')
        }
    ).rename('EVI');
    return evi;
};
