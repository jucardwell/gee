4.2.1 Filter by a Date Range

// filter by a date
var filtered_collection = image_collection.filterDate('2021-08-01', '2021-08-02');

//filter by a set of dates
var filtered_collection = image_collection.filterDate('2021-08-01', '2022-08-02');

//filter by a year or set of years
var filtered_collection = image_collection.filter(
    ee.Filter.calendarRange(2021, 2023, 'year')
);

//filter by a month
var filtered_collection = image_collection.filter(
    ee.Filter.calendarRange(8, 8, 'month')
);

//filter by day of year
var filtered_collection = image_collection.filter(
    ee.Filter.dayOfYear(150, 200)
);

//filter by day of the month

    ee.Filter.calendarRange(1, 15, 'day_of_month')
);

4.2.2 Filter by a Location

Filter an image collection to images that intersect a geometry.

// Filter by a point (coordinates)
var point = ee.Geometry.Point([-122.292, 37.901]);

var filtered_collection = image_collection.filterBounds(point);
// Filter by an ee.Geometry, ee.Feature, or ee.FeatureCollection
//in this case roi needs to be an ee.Geometry or ee.FeatureCollection

4.2.3 Filter by Metadata Properties

For these, you need to reference the actual metadata to see what the variables are called

//cloud cover less than 10%
var filtered_collection = image_collection.filter(
    ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 10)
);

//cloud cover greater than 10%
var filtered_collection = image_collection.filter(
    ee.Filter.gt('CLOUDY_PIXEL_PERCENTAGE', 10)
);


//cloud cover equal to 10%
var filtered_collection = image_collection.filter(
    ee.Filter.eq('CLOUDY_PIXEL_PERCENTAGE', 10)
);

//cloud cover between 10% and 20%
var filtered_collection = image_collection.filter(
    ee.Filter.and(
        ee.Filter.gte('CLOUDY_PIXEL_PERCENTAGE', 10),
        ee.Filter.lte('CLOUDY_PIXEL_PERCENTAGE', 20)
    )
);

4.3.3 Visualizing an Image Collection

The Map.addLayer command expects a single image. To visualize an image collection you need to reduce it to a single image. The easiest way to do this is to take the median value for each pixel across the image collection. You can also use mean(), max(), or min() to reduce the image collection.


Map.addLayer(
    image_collection.median(), //takes the median of the image collection
    {
        bands: ['SR_B3', 'SR_B2', 'SR_B1'], //bands will automatically fill RGB
        min: min,
        max: max
    },
    'Layer Name' //name to show in Layer Manager
);

4.3.4 Help Picking Min/Max

An easy way to pick min/max values is to take a sample of the raster and use that to see percentile values


// for an image
var stats = image.select('band')
  .sample({
    region: image.geometry(),
    numPixels: 1000,
    scale: 30
  })
  .reduceColumns({
    reducer: ee.Reducer.percentile([2, 98]),
    selectors: ['band']
  });

print(stats);


// for an image collection
var stats = image_collection.median().select('band')
  .sample({
    region: image_collection.first().geometry(),
    numPixels: 1000,
    scale: 30

  .reduceColumns({
    reducer: ee.Reducer.percentile([2, 98]),

print(stats);

4.5 Scale and Offset

Many datasets store values as integers to save space. Before any analysis, convert them back with the scale (and offset, if there is one). Every dataset lists its scale and offset on its catalog page, and nowhere in the image itself.

// Sentinel-2 SR Harmonized: divide by 10000
var scaled = image.divide(10000);

// Landsat 8/9 Collection 2 Level 2: multiply by 0.0000275, then add -0.2
// this only scales the surface reflectance bands (SR_B1 to SR_B7)
function scaleLandsat(image) {
  var scaled = image.select('SR_B.').multiply(0.0000275).add(-0.2);
  return image.addBands(scaled, null, true);
}

var scaled_collection = image_collection.map(scaleLandsat);

// MODIS LST (MOD11A2): multiply by 0.02 to get Kelvin, subtract 273.15 for Celsius
var celsius = image.select('LST_Day_1km').multiply(0.02).subtract(273.15);

4.6 Band Math

Band math allows you to create new raster bands by mathematically combining existing bands. Common operations include addition, subtraction, multiplication, division, and normalized differences.

4.6.1 Basic Mathematical Operations

// select individual bands 

var band_1 = image.select('band_1'); 
var band_2 = image.select('band_2'); 

// add two bands 
var added = band_1.add(band_2); 

// subtract
var subtracted = band_1.subtract(band_2);

// multiply two bands 
var multiplied = band_1.multiply(band_2); 

// divide 
var divided = band_1.divide(band_2);

//built in normalized difference function

var normalized_difference = image.normalizedDifference(['band_1', 'band_2'])

4.6.2 Using Expressions

Expressions can be used for more complicated operations


var calculated_band = image.expression( 
                    '(BAND_1 - BAND_2) / (BAND_1 + BAND_2)', 
                    { BAND_1: image.select('band_1'), 
                    BAND_2: image.select('band_2') });

4.6.3 Add a Band to an Image

// calculate a new band and give it a name
var ndvi = image.normalizedDifference(['band_1', 'band_2']).rename('NDVI');

// add it to the original image
var with_ndvi = image.addBands(ndvi);

4.7.4 Cloud Masking


//Landsat 8/9 Collection 2 Level 2 
function maskLandsatClouds(image) {
  // Select the QA band that stores per-pixel quality flags
  var qa = image.select('QA_PIXEL');
  // Bit 1 = dilated cloud, 2 = cirrus, 3 = cloud, 4 = cloud shadow
  var badBits = (1 << 1) | (1 << 2) | (1 << 3) | (1 << 4);
  // A pixel is clear if NONE of those bits are turned on
  var clear = qa.bitwiseAnd(badBits).eq(0);
  // Keep the clear pixels, drop everything else
  return image.updateMask(clear);
}

// Sentinel-2 SR Harmonized, masked with Cloud Score+
function maskS2clouds(image) {
  // Look up this image's matching Cloud Score+ image by its ID
  var csPlus = ee.ImageCollection('GOOGLE/CLOUD_SCORE_PLUS/V1/S2_HARMONIZED');
  var match = csPlus.filter(
    ee.Filter.eq('system:index', image.get('system:index'))).first();
  // Scores run 0 (occluded) to 1 (clear)
  var score = ee.Image(match).select('cs_cdf');
  // A pixel is clear if its score is at or above the threshold
  var clear = score.gte(0.60);
  // Keep the clear pixels, drop everything else
  return image.updateMask(clear);
}

4.10 Reducers

4.10.1 Reduce an Image Within One Region

reduceRegions() (above) summarizes into every feature in a feature collection. reduceRegion() summarizes into a single geometry and returns a dictionary.

var proj = image.projection(); // if reducing a raw image, use the image itself
// var proj = COLLECTION.first().projection(); // if reducing a composite, use the original collection instead

var stats = image.reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: roi,
    scale: proj.nominalScale(),
    crs: proj,
    maxPixels: 1e9
});

print(stats);

4.10.3 Composite Categories with the Mode

For categorical data like land cover, a median or mean would invent classes that don’t exist. Use the mode, the most common value at each pixel.

4.11 Classification

4.11.1 Supervised Classification

Draw training polygons with the geometry tools. Import each class as a FeatureCollection with a property called ‘class’ (1, 2, 3, …).

// combine the training classes into one feature collection
var training_features = ee.FeatureCollection([
    class_1, class_2, class_3, class_4
]).flatten();

// the bands the classifier will use
var prediction_bands = ['band_1', 'band_2', 'band_3'];

// pull the band values from inside the training polygons
var training = image.select(prediction_bands).sampleRegions({
    collection: training_features,
    properties: ['class'],
    scale: 10
});

// train a classifier (CART)
var classifier = ee.Classifier.smileCart().train({
    features: training,
    classProperty: 'class',
    inputProperties: prediction_bands
});

// or a random forest with 50 trees
// var classifier = ee.Classifier.smileRandomForest(50).train({ ...same as above... });

// classify the image
var classified = image.select(prediction_bands).classify(classifier);

Map.addLayer(classified, {min: 1, max: 4, palette: ['blue', 'red', 'green', 'orange']}, 'Classified');

4.11.2 Unsupervised Classification (k-means)

// take a random sample of pixels
var sample = image.select(prediction_bands).sample({
    region: roi,
    scale: 10,
    numPixels: 1000
});

// group the pixels into 4 clusters
var clusterer = ee.Clusterer.wekaKMeans(4).train(sample);
var clustered = image.select(prediction_bands).cluster(clusterer);

Map.addLayer(clustered.randomVisualizer(), {}, 'K-means clusters');

4.12 Regression

4.12.1 Linear Regression

var proj = image_collection.first().projection();

// the image needs a band of 1s (for the intercept), then the predictors, then the response
var training_image = ee.Image(1)
    .addBands(predictors)
    .addBands(response);

// numX = number of predictor bands + 1 (for the band of 1s)
var regression = training_image.reduceRegion({
    reducer: ee.Reducer.linearRegression({numX: 7, numY: 1}),
    geometry: roi,
    scale: proj.nominalScale(),
    crs: proj,
    maxPixels: 1e9
});

// turn the coefficients into a list
var coefficients = ee.Array(regression.get('coefficients'))
    .project([0])
    .toList();

// predict: multiply each band by its coefficient and add them up
var predicted = ee.Image(1)
    .addBands(predictors)
    .multiply(ee.Image.constant(coefficients))
    .reduce(ee.Reducer.sum())
    .rename('prediction');

4.12.2 Random Forest Regression

// sample pixels from an image with the predictor bands and the response band
var training = predictors.addBands(response).sample({
    region: roi,
    scale: 30,
    numPixels: 1500,
    seed: 5
});

var rf = ee.Classifier.smileRandomForest(50)
    .setOutputMode('REGRESSION')
    .train({
        features: training,
        classProperty: 'response_band_name',
        inputProperties: prediction_bands
    });

var predicted = predictors.classify(rf).rename('prediction');

4.12.3 Root Mean Squared Error

var squared_error = predicted.subtract(response).pow(2).rename('squared_error');

var mse = squared_error.reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: roi,
    scale: 30,
    maxPixels: 1e9
}).get('squared_error');

print('RMSE', ee.Number(mse).sqrt());

4.13.3 Linear Trend

// each image needs two bands, in this order: year, then the value
var annual = ee.ImageCollection.fromImages(years.map(function(y) {
    var start = ee.Date.fromYMD(y, 1, 1);
    var value = image_collection.filterDate(start, start.advance(1, 'year')).median();
    var year_band = ee.Image.constant(y).rename('year').float();
    return year_band.addBands(value).set('system:time_start', start.millis());
}));

// 'scale' is the slope (change per year), 'offset' is the intercept
var trend = annual.reduce(ee.Reducer.linearFit());

Map.addLayer(trend.select('scale'), {min: -0.5, max: 0.5, palette: ['blue', 'white', 'red']}, 'Trend');

4.13.4 Anomaly

// the period of interest
var period = image_collection.filterDate('2024-09-01', '2024-10-01').sum();

// the baseline: the same month across many years (here 30 Septembers)
var normal = image_collection
    .filterDate('1994-01-01', '2024-01-01')
    .filter(ee.Filter.calendarRange(9, 9, 'month'))
    .sum()
    .divide(30);

// difference from normal (original units)
var difference = period.subtract(normal);

// percent of normal (100 = normal)
var percent = period.divide(normal).multiply(100);

4.14 Change Detection

4.14.1 Image Differencing

// same index, same season, two years
var before_index = before.normalizedDifference(['band_1', 'band_2']);
var after_index = after.normalizedDifference(['band_1', 'band_2']);

// before minus after: a loss comes out positive
var difference = before_index.subtract(after_index);

// threshold the difference to separate change from noise
var changed = difference.gte(threshold);

4.14.2 Post-Classification Comparison

// classify (or composite) each date into land cover
var lc_before = image_collection.filterDate('2016-01-01', '2017-01-01').mode();
var lc_after = image_collection.filterDate('2025-01-01', '2026-01-01').mode();

// transition code: tens digit is "from", ones digit is "to"
var transitions = lc_before.multiply(10).add(lc_after);

// pixels that went from class 1 to class 6
var one_to_six = transitions.eq(16);

Map.addLayer(one_to_six.updateMask(one_to_six), {palette: ['red']}, 'Class 1 to class 6');

// Sentinel-2 SR Harmonized, masked with Cloud Score+
function maskS2clouds(image) {
  // Look up this image's matching Cloud Score+ image by its ID
  var csPlus = ee.ImageCollection('GOOGLE/CLOUD_SCORE_PLUS/V1/S2_HARMONIZED');
  var match = csPlus.filter(
    ee.Filter.eq('system:index', image.get('system:index'))).first();
  // Scores run 0 (occluded) to 1 (clear)
  var score = ee.Image(match).select('cs_cdf');
  // A pixel is clear if its score is at or above the threshold
  var clear = score.gte(0.60);
  // Keep the clear pixels, drop everything else
  return image.updateMask(clear);
}

//I chose the median because it takes the middle value at each pixel, so any
//clouds or shadows that the mask missed (which are unusually bright or dark)
//will not pull the composite the way they would with the mean. Max would
//favor leftover clouds and min would favor shadows.

//filter data, apply cloud mask, take median, clip, scale
var pre = ee.ImageCollection("COPERNICUS/S2_SR_HARMONIZED")
                            .filterBounds(studyarea)
                            .filterDate('2019-07-01', '2019-09-01')
                            .map(maskS2clouds)
                            .median()
                            .clip(studyarea)
                            .divide(10000);
