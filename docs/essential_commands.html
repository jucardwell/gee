2  Essential Commands

2.1 Accessing Images and Image Collections

2.1.1 Accessing an Image and Viewing Information

// add a single image
var image = ee.Image("image_name");

//this will print information about the image in the console
print(image);

2.1.2 Accessing an Image Collection and Viewing Information

// add an image collection
var image_collection = ee.ImageCollection("image_collection_name");

//this will print information about the first image 
//in the image collection in the console
print(image_collection.first());

2.2 Filtering Image Collections

2.2.1 Filter by a Date Range

// filter by a date
var filtered_collection = image_collection.filterDate('2021-08-01', '2021-08-02');

//filter by a set of dates
var filtered_collection = image_collection.filterDate('2021-08-01', '2022-08-02');

//filter by a year or set of years
var filtered_collection = image_collection.filter(
    ee.Filter.calendarRange(2021, 2023, 'year')
);

//filter by a month
var filtered_collection = image_collection.filter(
    ee.Filter.calendarRange(8, 8, 'month')
);

//filter by day of year
var filtered_collection = image_collection.filter(
    ee.Filter.dayOfYear(150, 200)
);

//filter by day of the month
var filtered_collection = image_collection.filter(
    ee.Filter.calendarRange(1, 15, 'day_of_month')
);

2.2.2 Filter by a Location

Filter an image collection to images that intersect a geometry.

// Filter by a point (coordinates)
var point = ee.Geometry.Point([-122.292, 37.901]);

var filtered_collection = image_collection.filterBounds(point);
// Filter by an ee.Geometry, ee.Feature, or ee.FeatureCollection
//in this case roi needs to be an ee.Geometry or ee.FeatureCollection
var filtered_collection = image_collection.filterBounds(roi);

2.2.3 Filter by Metadata Properties

For these, you need to reference the actual metadata to see what the variables are called

//cloud cover less than 10%
var filtered_collection = image_collection.filter(
    ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 10)
);

//cloud cover greater than 10%
var filtered_collection = image_collection.filter(
    ee.Filter.gt('CLOUDY_PIXEL_PERCENTAGE', 10)
);


//cloud cover equal to 10%
var filtered_collection = image_collection.filter(
    ee.Filter.eq('CLOUDY_PIXEL_PERCENTAGE', 10)
);

//cloud cover between 10% and 20%
var filtered_collection = image_collection.filter(
    ee.Filter.and(
        ee.Filter.gte('CLOUDY_PIXEL_PERCENTAGE', 10),
        ee.Filter.lte('CLOUDY_PIXEL_PERCENTAGE', 20)
    )
);

2.3 Visualization

2.3.1 Visualizing a Single-Band


Map.addLayer(
    image, //  image to display
    {
        bands: ['band_name'], //  band to display
        min: min, //  display range  
        max: max
    },
    'Layer Name' //  name to show in Layer Manager
);

2.3.2 Visualizing a Multi-Band


Map.addLayer(
    image, //image to display
    {
        bands: ['SR_B3', 'SR_B2', 'SR_B1'], //bands will automatically fill RGB
        min: min,
        max: max
    },
    'Layer Name' //name to show in Layer Manager
);

2.3.3 Visualizing an Image Collection

The Map.addLayer command expects a single image. To visualize an image collection you need to reduce it to a single image. The easiest way to do this is to take the median value for each pixel across the image collection. You can also use mean(), max(), or min() to reduce the image collection.


Map.addLayer(
    image_collection.median(), //takes the median of the image collection
    {
        bands: ['SR_B3', 'SR_B2', 'SR_B1'], //bands will automatically fill RGB
        min: min,
        max: max
    },
    'Layer Name' //name to show in Layer Manager
);

2.3.4 Help Picking Min/Max

An easy way to pick min/max values is to take a sample of the raster and use that to see percentile values


// for an image
var stats = image.select('band')
  .sample({
    region: image.geometry(),
    numPixels: 1000,
    scale: 30
  })
  .reduceColumns({
    reducer: ee.Reducer.percentile([2, 98]),
    selectors: ['band']
  });

print(stats);


// for an image collection
var stats = image_collection.median().select('band')
  .sample({
    region: image_collection.first().geometry(),
    numPixels: 1000,
    scale: 30
  })
  .reduceColumns({
    reducer: ee.Reducer.percentile([2, 98]),
    selectors: ['band']
  });

print(stats);

2.3.5 Setting Color Palette for Single-Band Image

Start by adding this repo to your GEE account. You can view all the palette options here


var palettes = require('users/gena/packages:palettes');

Map.addLayer(
    image, //  image to display
    {
        bands: ['band_name'], //  band to display
        min: min, //  display range  
        max: max,
        palette: palettes.colorbrewer.RdYlGn[9] //set palette and number of classes
    },
    'Layer Name' //  name to show in Layer Manager
);

Or you can manually set palettes:


Map.addLayer(
    image, //  image to display
    {
        bands: ['band_name'], //  band to display
        min: min, //  display range  
        max: max,
        palette: ['blue', 'white', 'green'] //set palette manually
    },
    'Layer Name' //  name to show in Layer Manager
);

2.5 Band Math

Band math allows you to create new raster bands by mathematically combining existing bands. Common operations include addition, subtraction, multiplication, division, and normalized differences.

2.5.1 Basic Mathematical Operations

// select individual bands 

var band_1 = image.select('band_1'); 
var band_2 = image.select('band_2'); 

// add two bands 
var added = band_1.add(band_2); 

// subtract
var subtracted = band_1.subtract(band_2);

// multiply two bands 
var multiplied = band_1.multiply(band_2); 

// divide 
var divided = band_1.divide(band_2);

//built in normalized difference function

var normalized_difference = image.normalizedDifference(['band_1', 'band_2'])

2.5.2 Using Expressions

Expressions can be used for more complicated operations


var calculated_band = image.expression( 
                    '(BAND_1 - BAND_2) / (BAND_1 + BAND_2)', 
                    { BAND_1: image.select('band_1'), 
                    BAND_2: image.select('band_2') });

2.6 Thresholds, Reclassifying, and Masking

2.6.1 Threshold


// greater than a threshold
var thresholded = image.gt(threshold);

// greater than or equal to a threshold
var thresholded = image.gte(threshold);

// less than a threshold
var thresholded = image.lt(threshold);

// less than or equal to a threshold
var thresholded = image.lte(threshold);

// equal to a threshold
var thresholded = image.eq(threshold);

2.6.2 Reclassifying


// create multiple classes
var reclassified = image
    .where(image.lt(threshold_1), value_1)
    .where(
        image.gte(threshold_1)
            .and(image.lt(threshold_2)),
        value_2
    )
    .where(image.gte(threshold_2), value_3);

2.6.3 Masking

// create a mask
var mask = image.gt(threshold);

// apply the mask
var masked_image = image.updateMask(mask);

2.6.4 Clipping Images


var clipped = image.clip(roi);

var projection = image.projection();

var output = image.reduceRegions({

});

2.9 Exporting Data

2.9.1 Export an Image


//get the image projection
var projection = image.projection();

//get the pixel size
var scale = projection.nominalScale();

//export the image
Export.image.toDrive({
    image: image, //image to export
    description: 'File_Name', //name of the exported file
    scale: scale, //pixel size
    fileFormat: 'GeoTIFF' //output format
});

2.9.2 Export a Feature Collection

Export.table.toDrive({
    collection: feature_collection,
    description: 'File_Name',
    fileFormat: 'CSV'
});
